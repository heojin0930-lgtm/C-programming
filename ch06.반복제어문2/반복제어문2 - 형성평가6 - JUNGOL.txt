문제
두 개의 정수를 입력받아 작은 수부터 큰 수까지 3의 배수이거나 5의 배수인 수들의 합과 평균을 출력하는 프로그램을 작성하시오.

#include <stdio.h>
int main(void) {
int a, b, i;
int min, max;
int sum = 0;
int count = 0;
double ave = 0;
scanf("%d %d", &a, &b);
if (a < b) {
min = a;
max = b;
}
else {
min = b;
max = a;
}
for (i = min; i <= max; i++) {
if (i % 3 == 0 || i % 5 == 0) {
sum += i;
count++;
}
}
if (count > 0) {
ave = (double)sum / count;
}
printf("%d\n", sum);
printf("%.1f\n", ave);
return 0;
}
